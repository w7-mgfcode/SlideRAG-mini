### Section 3: Workflow Shape

workflow_type: "iterative-loop"
phases:
  - name: "Bootstrap"
    description: "Set up workspace and clarify research question"
    inputs: "User brief or question"
    outputs: "Workspace dirs + research-state.yaml"
    validation: "State file exists and passes schema validation"
  - name: "Survey"
    description: "Gather literature and prior work"
    inputs: "Search terms, web access"
    outputs: "Literature notes in literature/"
    validation: "5+ sources, 2+ per key claim"
  - name: "Hypothesize"
    description: "Form testable hypotheses"
    inputs: "Literature survey results"
    outputs: "2-5 hypotheses in state"
    validation: "Each hypothesis is falsifiable with defined test method"
  - name: "Experiment"
    description: "Run experiments for each hypothesis"
    inputs: "Hypothesis + experiment design"
    outputs: "Results in experiments/"
    validation: "Success criteria evaluated for each experiment"
  - name: "Synthesize"
    description: "Integrate results, identify gaps"
    inputs: "All experiment results + literature"
    outputs: "Synthesis doc + HTML presentation"
    validation: "All hypotheses resolved or gaps documented"
  - name: "Write"
    description: "Produce final research report"
    inputs: "All findings and synthesis docs"
    outputs: "findings/report.md + final presentation"
    validation: "Report covers all findings with evidence chains"
